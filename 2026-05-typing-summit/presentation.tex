\begin{frame}[fragile]
    \frametitle{Constraint sets are sets of typevar assignments}

    \begin{uncoverenv}<2->
    \begin{displaymath}
        \cset{\pytype{Literal[True]} \preceq \typevar{T}} =
        \left\{
            \begin{array}{l}
                \assignment{T}{bool} \\
                \assignment{T}{Literal[True]} \\
                \assignment{T}{int} \\
                \assignment{T}{object} \\
                \assignment{T}{SupportsIndex} \\
                \multicolumn{1}{c}{\vdots} \\
            \end{array}
        \right\}
    \end{displaymath}
    \end{uncoverenv}

    \vskip 4ex

    \begin{uncoverenv}<3->
    \begin{displaymath}
        \cset{
            (\pytype{Literal[True]} \preceq \typevar{T}) \wedge
            (\typevar{T} \preceq \pytype{bool})
        } =
        \left\{
            \begin{array}{l}
                \assignment{T}{bool} \\
                \assignment{T}{Literal[True]} \\
            \end{array}
        \right\}
    \end{displaymath}
    \end{uncoverenv}
\end{frame}
